# TryZore - Unified Master Product & Engineering Blueprint

**Version 18.0 - Production-Ready Convex Architecture with Resilience Engineering**

## Document Purpose

This is the definitive, production-ready blueprint for the TryZore platform built on Convex.dev architecture. It combines Convex's simplicity and real-time capabilities with comprehensive resilience patterns, error handling, and recovery mechanisms required for production systems. This document serves as the single source of truth for building a robust, scalable fitness platform.

## Critical Architecture Principles

Before diving into implementation, these principles govern all technical decisions:

1. **Defensive Programming**: Every external API call must assume failure
2. **Graceful Degradation**: The system remains functional when third-party services fail
3. **Audit Everything**: All state changes are logged for debugging and compliance
4. **Recovery Over Prevention**: Design for quick recovery from failures, not just prevention
5. **User Experience First**: Technical failures should never leave users in ambiguous states

### 1.2 External Service Contracts with Resilience

External services are wrapped in resilient action patterns that handle failures gracefully:

#### Resilient Action Pattern

```typescript
// convex/lib/resilientAction.ts
import { ActionCtx } from "../_generated/server";
import { Id } from "../_generated/dataModel";
import { internal } from "../_generated/api";

export interface ResilientActionConfig {
  service: "kisi" | "stripe" | "email" | "sms";
  operation: string;
  maxAttempts?: number;
  backoffMultiplier?: number;
  priority?: number;
  idempotencyKey?: string;
}

export class ResilientAction {
  constructor(private config: ResilientActionConfig) {}

  async execute(
    ctx: ActionCtx,
    payload: any,
    immediate: boolean = false
  ): Promise<any> {
    // Check circuit breaker
    const circuitBreaker = await ctx.runQuery(
      internal.circuitBreaker.getState,
      {
        service: this.config.service,
        endpoint: this.config.operation,
      }
    );

    if (circuitBreaker.state === "open") {
      // Circuit is open, queue for later
      await this.queueForRetry(ctx, payload);
      throw new Error(`Service ${this.config.service} is currently unavailable`);
    }

    if (immediate) {
      // Try immediate execution with circuit breaker protection
      try {
        const result = await this.executeWithCircuitBreaker(ctx, payload);
        await this.recordSuccess(ctx);
        return result;
      } catch (error) {
        await this.recordFailure(ctx, error);
        
        // Queue for retry if not a permanent failure
        if (!this.isPermanentFailure(error)) {
          await this.queueForRetry(ctx, payload);
        }
        
        throw error;
      }
    } else {
      // Queue for background processing
      await this.queueForRetry(ctx, payload, 0);
      return { queued: true, message: "Operation queued for processing" };
    }
  }

  private async queueForRetry(
    ctx: ActionCtx,
    payload: any,
    delayMs: number = 0
  ) {
    const nextAttemptAt = Date.now() + delayMs;
    
    await ctx.runMutation(internal.apiQueue.enqueue, {
      service: this.config.service,
      operation: this.config.operation,
      payload,
      priority: this.config.priority || 5,
      maxAttempts: this.config.maxAttempts || 5,
      nextAttemptAt,
      idempotencyKey: this.config.idempotencyKey,
    });
  }

  private calculateBackoff(attemptNumber: number): number {
    const baseDelay = 1000; // 1 second
    const multiplier = this.config.backoffMultiplier || 2;
    const maxDelay = 300000; // 5 minutes
    
    const delay = Math.min(
      baseDelay * Math.pow(multiplier, attemptNumber),
      maxDelay
    );
    
    // Add jitter to prevent thundering herd
    return delay + Math.random() * 1000;
  }
  
  private isPermanentFailure(error: any): boolean {
    // Define permanent failures that shouldn't be retried
    const permanentErrorCodes = [
      'invalid_api_key',
      'subscription_canceled',
      'invalid_request',
      'resource_not_found',
    ];
    
    return permanentErrorCodes.includes(error.code);
  }
}
```

### 1.3 Permission System Architecture with Audit Trail

Every permission check is logged for security and compliance:

```typescript
// convex/lib/permissions.ts
import { QueryCtx, MutationCtx } from "../_generated/server";
import { Id } from "../_generated/dataModel";

export async function requireAuth(
  ctx: QueryCtx | MutationCtx,
  operation?: string
): Promise<Id<"users">> {
  const identity = await ctx.auth.getUserIdentity();
  if (!identity) {
    // Log failed auth attempt
    if ('db' in ctx) {
      await ctx.db.insert("auditLog", {
        actorUserId: undefined,
        action: "auth_failed",
        targetEntityId: operation,
        details: { 
          operation,
          timestamp: Date.now(),
          reason: "no_identity"
        },
      });
    }
    throw new Error("Unauthorized: No authenticated user");
  }
  
  const user = await ctx.db
    .query("users")
    .withIndex("by_workos", q => q.eq("workosUserId", identity.subject))
    .unique();
    
  if (!user) {
    if ('db' in ctx) {
      await ctx.db.insert("auditLog", {
        actorUserId: undefined,
        action: "auth_failed",
        targetEntityId: identity.subject,
        details: { 
          operation,
          timestamp: Date.now(),
          reason: "user_not_found"
        },
      });
    }
    throw new Error("User not found");
  }
  
  return user._id;
}

export async function requireOrgAccess(
  ctx: QueryCtx | MutationCtx,
  orgId: Id<"organizations">,
  requiredPermission?: string
): Promise<void> {
  const userId = await requireAuth(ctx);
  
  // Check if user has access to this organization
  const assignment = await ctx.db
    .query("crossOrgStaffAssignments")
    .withIndex("by_user", q => q.eq("userId", userId))
    .filter(q => q.eq(q.field("organizationId"), orgId))
    .unique();
    
  if (!assignment) {
    // Log unauthorized access attempt
    if ('db' in ctx) {
      await ctx.db.insert("auditLog", {
        actorUserId: userId,
        action: "unauthorized_org_access",
        targetEntityId: orgId,
        details: { 
          requiredPermission,
          timestamp: Date.now()
        },
      });
    }
    throw new Error("Access denied: No access to this organization");
  }
  
  if (requiredPermission) {
    const hasPermission = await checkPermission(
      ctx,
      assignment.roleId,
      requiredPermission
    );
    
    if (!hasPermission) {
      // Log permission denial
      if ('db' in ctx) {
        await ctx.db.insert("auditLog", {
          actorUserId: userId,
          action: "permission_denied",
          targetEntityId: orgId,
          details: { 
            requiredPermission,
            roleId: assignment.roleId,
            timestamp: Date.now()
          },
        });
      }
      throw new Error(`Access denied: Missing permission ${requiredPermission}`);
    }
  }
  
  // Log successful access for sensitive operations
  if (requiredPermission?.includes('financial') || 
      requiredPermission?.includes('admin')) {
    if ('db' in ctx) {
      await ctx.db.insert("auditLog", {
        actorUserId: userId,
        action: "sensitive_access_granted",
        targetEntityId: orgId,
        details: { 
          permission: requiredPermission,
          timestamp: Date.now()
        },
      });
    }
  }
}
```

### 2.3 Multi-Step Onboarding Architecture with Recovery

The onboarding system is designed to handle all failure modes gracefully:

```typescript
// convex/onboarding.ts
import { mutation, query } from "./_generated/server";
import { v } from "convex/values";
import { Id } from "./_generated/dataModel";

// Recovery service for handling abandoned/failed onboarding
export const analyzeOnboardingState = query({
  args: {
    email: v.string(),
  },
  handler: async (ctx, args) => {
    // Check for existing owner account
    const ownerAccount = await ctx.db
      .query("ownerAccounts")
      .withIndex("by_email", q => q.eq("email", args.email))
      .unique();
      
    // Check for active sessions
    const activeSession = await ctx.db
      .query("onboardingSessions")
      .withIndex("by_owner", q => 
        ownerAccount ? q.eq("ownerAccountId", ownerAccount._id) : q.eq("ownerAccountId", undefined)
      )
      .filter(q => q.gt(q.field("expiresAt"), Date.now()))
      .order("desc")
      .first();
      
    // Check for organizations
    const organizations = ownerAccount ? 
      await ctx.db
        .query("ownerOrganizations")
        .withIndex("by_owner", q => q.eq("ownerAccountId", ownerAccount._id))
        .collect() : [];
        
    // Determine recovery action
    if (!ownerAccount) {
      return {
        action: "start_fresh",
        reason: "No existing account found",
      };
    }
    
    if (organizations.length > 0) {
      // Check if setup is complete
      const org = await ctx.db.get(organizations[0].organizationId);
      
      if (!org?.stripeAccountId) {
        return {
          action: "resume_stripe_setup",
          reason: "Organization exists but Stripe not connected",
          sessionToken: activeSession?.sessionToken,
          organizationId: organizations[0].organizationId,
        };
      }
      
      if (!org?.kisiApiKeyEncrypted) {
        return {
          action: "resume_kisi_setup",
          reason: "Organization exists but Kisi not configured",
          sessionToken: activeSession?.sessionToken,
          organizationId: organizations[0].organizationId,
        };
      }
      
      return {
        action: "login",
        reason: "Setup complete",
        organizationId: organizations[0].organizationId,
      };
    }
    
    if (activeSession) {
      // Analyze session state
      const hoursSinceLastUpdate = 
        (Date.now() - activeSession._creationTime) / (1000 * 60 * 60);
        
      if (hoursSinceLastUpdate < 1) {
        return {
          action: "resume_from_step",
          reason: "Recent session found",
          sessionToken: activeSession.sessionToken,
          currentStep: activeSession.currentStep,
        };
      } else if (hoursSinceLastUpdate < 24) {
        return {
          action: "offer_resume_or_restart",
          reason: "Session found but may be stale",
          sessionToken: activeSession.sessionToken,
          currentStep: activeSession.currentStep,
        };
      } else {
        return {
          action: "cleanup_and_retry",
          reason: "Session expired",
          sessionToken: activeSession.sessionToken,
        };
      }
    }
    
    // Orphaned account
    return {
      action: "cleanup_and_retry",
      reason: "Account exists but no organization or active session",
      requiresManualReview: true,
    };
  },
});

export const startOnboarding = mutation({
  args: {
    email: v.string(),
    name: v.string(),
  },
  handler: async (ctx, args) => {
    // Check for existing state
    const existingState = await analyzeOnboardingState(ctx, { email: args.email });
    
    if (existingState.action !== "start_fresh") {
      throw new Error(`Cannot start new onboarding: ${existingState.reason}`);
    }
    
    // Create owner account
    const ownerAccountId = await ctx.db.insert("ownerAccounts", {
      email: args.email,
      name: args.name,
      isEnterprise: false,
    });
    
    // Create session with comprehensive tracking
    const sessionToken = generateSecureToken();
    const sessionId = await ctx.db.insert("onboardingSessions", {
      sessionToken,
      ownerAccountId,
      currentStep: "email_verification_pending",
      stepData: {},
      stepHistory: [{
        step: "account_created",
        timestamp: Date.now(),
        success: true,
      }],
      expiresAt: Date.now() + 24 * 60 * 60 * 1000,
    });
    
    // Queue email verification
    await ctx.scheduler.runAfter(0,
      internal.onboarding.sendVerificationEmail,
      { ownerAccountId, sessionId }
    );
    
    return { sessionToken, ownerAccountId };
  },
});

export const resumeOnboarding = mutation({
  args: {
    sessionToken: v.string(),
  },
  handler: async (ctx, args) => {
    const session = await ctx.db
      .query("onboardingSessions")
      .withIndex("by_token", q => q.eq("sessionToken", args.sessionToken))
      .unique();
      
    if (!session) {
      throw new Error("Invalid session");
    }
    
    if (session.expiresAt < Date.now()) {
      // Try to recover
      const newToken = generateSecureToken();
      const newSessionId = await ctx.db.insert("onboardingSessions", {
        sessionToken: newToken,
        ownerAccountId: session.ownerAccountId,
        currentStep: session.currentStep,
        stepData: session.stepData,
        stepHistory: [
          ...session.stepHistory,
          {
            step: "session_recovered",
            timestamp: Date.now(),
            success: true,
          },
        ],
        recoveryState: {
          attemptCount: (session.recoveryState?.attemptCount || 0) + 1,
          lastAttemptAt: Date.now(),
          suggestedAction: "resume",
        },
        expiresAt: Date.now() + 24 * 60 * 60 * 1000,
      });
      
      return {
        sessionToken: newToken,
        currentStep: session.currentStep,
        recovered: true,
      };
    }
    
    return {
      sessionToken: session.sessionToken,
      currentStep: session.currentStep,
      stepData: session.stepData,
      recovered: false,
    };
  },
});

export const completeOnboardingStep = mutation({
  args: {
    sessionToken: v.string(),
    step: v.string(),
    data: v.any(),
  },
  handler: async (ctx, args) => {
    const session = await ctx.db
      .query("onboardingSessions")
      .withIndex("by_token", q => q.eq("sessionToken", args.sessionToken))
      .unique();
      
    if (!session || session.expiresAt < Date.now()) {
      throw new Error("Invalid or expired session");
    }
    
    try {
      // Execute step-specific logic
      let nextStep: string;
      let result: any;
      
      switch (args.step) {
        case "email_verification":
          // Verify email
          await ctx.db.patch(session.ownerAccountId!, {
            emailVerifiedAt: Date.now(),
          });
          nextStep = "organization_pending";
          break;
          
        case "create_organization":
          // Create org with proper error handling
          const orgId = await ctx.db.insert("organizations", {
            name: args.data.name,
            planTier: "free",
            jurisdiction: args.data.jurisdiction || "DEFAULT",
            isFranchiseLocation: false,
            stripeTaxConfigured: false,
            kisiSyncStatus: "synced", // No Kisi yet
            stripeSyncStatus: "synced", // No Stripe yet
          });
          
          await ctx.db.insert("ownerOrganizations", {
            ownerAccountId: session.ownerAccountId!,
            organizationId: orgId,
            role: "owner",
            addedAt: Date.now(),
          });
          
          result = { orgId };
          nextStep = "stripe_connect_pending";
          break;
          
        case "stripe_connect":
          // This will be handled by Stripe OAuth callback
          nextStep = "kisi_setup_pending";
          break;
          
        case "kisi_setup":
          // Optional - can be skipped
          nextStep = "complete";
          break;
          
        default:
          throw new Error(`Unknown step: ${args.step}`);
      }
      
      // Update session
      await ctx.db.patch(session._id, {
        currentStep: nextStep,
        stepData: { ...session.stepData, [args.step]: args.data },
        stepHistory: [
```

```typescript
// Execute a single queue item
export const executeQueueItem = internalAction({
  args: {
    itemId: v.id("apiCallQueue"),
  },
  handler: async (ctx, args) => {
    const item = await ctx.runQuery(internal.apiQueue.getById, {
      id: args.itemId,
    });
    
    if (!item || item.status !== "processing") {
      return; // Already processed or doesn't exist
    }
    
    try {
      // Route to appropriate handler
      let result;
      switch (item.service) {
        case "kisi":
          result = await handleKisiOperation(ctx, item);
          break;
        case "stripe":
          result = await handleStripeOperation(ctx, item);
          break;
        case "email":
          result = await handleEmailOperation(ctx, item);
          break;
        default:
          throw new Error(`Unknown service: ${item.service}`);
      }
      
      // Mark as completed
      await ctx.runMutation(internal.apiQueue.markCompleted, {
        itemId: args.itemId,
        result,
      });
      
    } catch (error: any) {
      // Check if it's a permanent failure
      const isPermanent = isPermanentError(error);
      
      if (isPermanent || item.attempts >= item.maxAttempts) {
        // Move to dead letter queue
        await ctx.runMutation(internal.apiQueue.moveToDeadLetter, {
          itemId: args.itemId,
          error: error.message,
          errorDetails: error,
        });
      } else {
        // Calculate next retry time with exponential backoff
        const backoffMs = calculateBackoff(item.attempts);
        
        await ctx.runMutation(internal.apiQueue.markFailed, {
          itemId: args.itemId,
          error: error.message,
          nextAttemptAt: Date.now() + backoffMs,
        });
      }
    }
  },
});

function calculateBackoff(attemptNumber: number): number {
  const baseDelay = 1000; // 1 second
  const maxDelay = 300000; // 5 minutes
  
  const delay = Math.min(
    baseDelay * Math.pow(2, attemptNumber),
    maxDelay
  );
  
  // Add jitter
  return delay + Math.random() * 1000;
}

function isPermanentError(error: any): boolean {
  const permanentCodes = [
    'PERMANENT_FAILURE',
    'INVALID_API_KEY',
    'RESOURCE_NOT_FOUND',
    'INVALID_REQUEST',
  ];
  
  return permanentCodes.some(code => 
    error.message?.includes(code)
  );
}
```

```typescript
      .query("circuitBreakers")

      return {
        state: "closed",
        canProceed: true,
      };
    }
    
    // Check if half-open should transition to closed
    if (breaker.state === "half_open") {
      const successRate = breaker.successCount / 
        (breaker.successCount + breaker.failureCount);
        
      if (successRate > 0.8 && breaker.successCount >= 5) {
        // Enough successes, close the circuit
        await ctx.db.patch(breaker._id, {
          state: "closed",
          failureCount: 0,
          successCount: 0,
        });
        
        return { state: "closed", canProceed: true };
      }
    }
    
    // Check if open should transition to half-open
    if (breaker.state === "open" && 
        breaker.nextRetryAt && 
        breaker.nextRetryAt <= Date.now()) {
      await ctx.db.patch(breaker._id, {
        state: "half_open",
        successCount: 0,
        failureCount: 0,
      });
      
      return { state: "half_open", canProceed: true };
    }
    
    return {
      state: breaker.state,
      canProceed: breaker.state !== "open",
      nextRetryAt: breaker.nextRetryAt,
    };
  },
});
```

```typescript
export const purgeDeadLetterItem = mutation({
  args: {
    itemId: v.id("apiCallQueue"),
    reason: v.string(),
  },
  handler: async (ctx, args) => {
    const item = await ctx.db.get(args.itemId);
    
    if (!item || item.status !== "dead_letter") {
      throw new Error("Item not found in dead letter queue");
    }
    
    // Archive the item before deletion
    await ctx.db.insert("archivedApiCalls", {
      originalId: args.itemId,
      ...item,
      purgedAt: Date.now(),
      purgedReason: args.reason,
      purgedBy: await requireAuth(ctx),
    });
    
    // Delete from queue
    await ctx.db.delete(args.itemId);
    
    return { success: true };
  },
});
```

```typescript
export const getSystemHealth = query({
  handler: async (ctx) => {
    const now = Date.now();
    const fiveMinutesAgo = now - 5 * 60 * 1000;
    
    // Check API queue health
    const queueStats = await ctx.db
      .query("apiCallQueue")
      .withIndex("by_status_priority")
      .collect();
      
    const queueHealth = {
      pending: queueStats.filter(i => i.status === "pending").length,
      processing: queueStats.filter(i => i.status === "processing").length,
      failed: queueStats.filter(i => i.status === "failed").length,
      deadLetter: queueStats.filter(i => i.status === "dead_letter").length,
      oldestPending: queueStats
        .filter(i => i.status === "pending")
        .sort((a, b) => a._creationTime - b._creationTime)[0],
    };
    
    // Check circuit breakers
    const circuitBreakers = await ctx.db
      .query("circuitBreakers")
      .withIndex("by_state", q => q.eq("state", "open"))
      .collect();
      
    // Check recent errors
    const recentErrors = await ctx.db
      .query("auditLog")
      .withIndex("by_created")
      .filter(q => 
        q.and(
          q.gt(q.field("_creationTime"), fiveMinutesAgo),
          q.eq(q.field("action"), "error")
        )
      )
      .take(10);
      
    // Check sync status
    const orgsNeedingReview = await ctx.db
      .query("organizations")
      .withIndex("by_sync_status", q => 
        q.eq("kisiSyncStatus", "manual_review_required")
      )
      .collect();
      
    // Calculate health score
    let healthScore = 100;
    
    if (queueHealth.pending > 100) healthScore -= 20;
    if (queueHealth.deadLetter > 10) healthScore -= 30;
    if (circuitBreakers.length > 0) healthScore -= 20;
    if (recentErrors.length > 5) healthScore -= 10;
    if (orgsNeedingReview.length > 0) healthScore -= 10;
    
    const status = 
      healthScore >= 80 ? "healthy" :
      healthScore >= 60 ? "degraded" : "critical";
    
    return {
      status,
      healthScore,
      queueHealth,
      openCircuitBreakers: circuitBreakers.map(cb => ({
        service: cb.service,
        endpoint: cb.endpoint,
        openedAt: cb.lastFailureAt,
      })),
      recentErrorCount: recentErrors.length,
      orgsNeedingReview: orgsNeedingReview.length,
      timestamp: now,
    };

export const getMetrics = query({
  args: {
    timeRange: v.union(
      v.literal("hour"),
      v.literal("day"),
      v.literal("week")
    ),
  },
  handler: async (ctx, args) => {
    const now = Date.now();
    const startTime = 
      args.timeRange === "hour" ? now - 60 * 60 * 1000 :
      args.timeRange === "day" ? now - 24 * 60 * 60 * 1000 :
      now - 7 * 24 * 60 * 60 * 1000;
      
    // API call success rate
    const apiCalls = await ctx.db
      .query("apiCallQueue")
      .filter(q => q.gt(q.field("_creationTime"), startTime))
      .collect();
      
    const successRate = apiCalls.filter(c => c.status === "completed").length / 
      (apiCalls.length || 1);
      
    // Door unlock latency
    const unlockEvents = await ctx.db
      .query("auditLog")
      .withIndex("by_created")
      .filter(q => 
        q.and(
          q.gt(q.field("_creationTime"), startTime),
          q.eq(q.field("action"), "door_unlock_success")
        )
      )
      .collect();
      
    const avgLatency = unlockEvents.reduce((sum, e) => 
      sum + (e.details?.latencyMs || 0), 0
    ) / (unlockEvents.length || 1);
    
    // Reconciliation accuracy
    const reconciliations = await ctx.db
      .query("reconciliationLog")
      .withIndex("by_started")
      .filter(q => q.gt(q.field("startedAt"), startTime))
      .collect();
      
    const ambiguousCaseRate = reconciliations.reduce((sum, r) => 
      sum + r.ambiguousCases.length, 0
    ) / (reconciliations.length || 1);
    
    return {
      apiSuccessRate: successRate,
      avgDoorUnlockLatencyMs: avgLatency,
      totalApiCalls: apiCalls.length,
      failedApiCalls: apiCalls.filter(c => c.status === "failed").length,
      deadLetterCount: apiCalls.filter(c => c.status === "dead_letter").length,
      reconciliationAccuracy: 1 - ambiguousCaseRate,
      totalReconciliations: reconciliations.length,
      timeRange: args.timeRange,
    };
  },
});
```
